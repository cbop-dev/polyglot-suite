## 1. Objective & Scope

This specification defines the core TypeScript interfaces and reactive state management architecture for ACR. It establishes:
1. **Corpus-Agnostic Text Representations**: Eliminating hardcoded `greek`, `english`, and `latin` properties in favor of uniform `TextVersion` arrays.
2. **Dynamic Visibility Management**: Replacing discrete boolean flags (`showGreek`, `showEnglish`, `showLatin`) with a reactive set of visible language/version IDs (`visibleLanguages: Set<string>`).
3. **Clean Runes-based State**: Implementing state with Svelte 5 `$state` and `$derived`.

## 2. Core TypeScript Interfaces

### 2.1 Tokens & Text Units
```ts
export interface TextToken {
    w: string;                       // Surface word / form
    l?: string;                      // Normalized lemma
    m?: string;                      // Morphological parse tag
    p?: string;                      // Part of speech
    lang?: string;                   // Token language (e.g. 'grc', 'hbo')
    type?: 'word' | 'punctuation' | 'citation' | 'apparatus';
    // Citation metadata if type === 'citation'
    book_code?: string;
    book_name?: string;
    bible_chapter?: number;
    bible_verse?: number;
    display?: string;
}

export interface TextUnit {
    id: number;
    version_id: string;              // Foreign key to corpora(id), e.g. 'lake_grc'
    work_code: string;               // e.g. '1CLE'
    book: number;
    chapter: number;
    section: number;
    section_label?: string;          // Optional custom label (e.g. '1a')
    display_citation: string;        // e.g. '1 Clem. 1:1'
    milestone_ref?: string;          // e.g. '1', '15', 'M. 2.45'
    text_content: string;
    tokens?: TextToken[];
}
```

### 2.2 Generic Text Version (`TextVersion`)
A `TextVersion` packages all text units for a specific version/language within a division:
```ts
export interface TextVersion {
    versionId: string;               // e.g. 'lake_grc', 'lake_eng', 'af_lat'
    language: string;                // e.g. 'grc', 'eng', 'lat'
    label: string;                   // Full header label (e.g. 'Kirsopp Lake Greek Critical Text')
    shortLabel: string;              // Short badge label (e.g. 'Lake (1912)')
    direction: 'ltr' | 'rtl';
    fontClass: string;               // Resolved CSS classes (e.g. 'font-greek greek')
    textUnits: TextUnit[];
}

export interface DivisionResult {
    corpusId: string;
    workCode: string;
    division: number;
    textVersions: TextVersion[];
}
```

## 3. Reactive Reader State Architecture (`ReaderState`)

### 3.1 Store Contract (`src/lib/stores/readerState.svelte.ts`)
```ts
import { SvelteSet } from 'svelte/reactivity';
import { CORPORA_CONFIG, type CorpusConfig } from '../config/corpora.js';
import { getCorpusWorks, findWork, type CorpusWork } from '../config/corpora.data.js';
import type { TextToken } from './types.js';

export type ViewLayoutMode = 'parallel' | 'stacked' | 'single';

export class ReaderState {
    // Current Navigation Coordinates
    currentCorpus = $state<string>('af');
    currentWorkCode = $state<string>('1CLE');
    currentDivision = $state<number>(1);
    currentSection = $state<string | number | null>(null);

    // Layout & Language Visibility
    viewMode = $state<ViewLayoutMode>('parallel');
    visibleLanguages = $state<SvelteSet<string>>(new SvelteSet(['grc', 'eng']));

    // Side Panel & Modal States
    selectedToken = $state<TextToken | null>(null);
    selectedCitation = $state<string | null>(null);
    selectedScriptureCitation = $state<{
        bookCode?: string;
        chapter?: number;
        verse?: number;
        display?: string;
        bookName?: string;
    } | null>(null);

    isSidePanelOpen = $state<boolean>(false);
    activeSidePanelTab = $state<string>('lexeme');

    isCorpusModalOpen = $state<boolean>(false);
    isWorkModalOpen = $state<boolean>(false);
    isDivisionModalOpen = $state<boolean>(false);
    showHotkeyHelp = $state<boolean>(false);
    showCitationModal = $state<boolean>(false);

    // Reactive Derived Properties
    get currentCorpusConfig(): CorpusConfig {
        return CORPORA_CONFIG[this.currentCorpus] || CORPORA_CONFIG['af'];
    }

    get currentWork(): CorpusWork {
        const found = findWork(this.currentCorpus, this.currentWorkCode);
        return found || getCorpusWorks(this.currentCorpus)[0];
    }

    // Dynamic Language Visibility Controls
    isLanguageVisible(langOrVersionId: string): boolean {
        return this.visibleLanguages.has(langOrVersionId);
    }

    toggleLanguage(langOrVersionId: string) {
        if (this.visibleLanguages.has(langOrVersionId)) {
            // Guard against closing all columns
            if (this.visibleLanguages.size > 1) {
                this.visibleLanguages.delete(langOrVersionId);
            }
        } else {
            this.visibleLanguages.add(langOrVersionId);
        }
    }

    setVisibleLanguages(languages: string[]) {
        this.visibleLanguages.clear();
        for (const l of languages) {
            this.visibleLanguages.add(l);
        }
    }

    // Navigation Steppers
    setCorpus(corpusId: string, workCode?: string, division?: number, section?: string | number | null) {
        this.closeSidePanel();
        this.currentCorpus = corpusId;
        const cfg = this.currentCorpusConfig;
        
        // Reset visible languages to corpus defaults
        const defaults = cfg.textVersions
            .filter((v) => v.isDefaultVisible)
            .map((v) => v.language);
        this.setVisibleLanguages(defaults.length > 0 ? defaults : [cfg.primaryLanguage]);

        const works = getCorpusWorks(corpusId);
        const targetWork = workCode ? (findWork(corpusId, workCode) || works[0]) : works[0];
        this.currentWorkCode = targetWork.code;
        this.currentDivision = division !== undefined
            ? Math.max(targetWork.minDivision, Math.min(division, targetWork.totalDivisions))
            : targetWork.minDivision;
        this.currentSection = section ?? null;
    }

    setWork(workCode: string, division?: number, section?: string | number | null) {
        const target = findWork(this.currentCorpus, workCode);
        if (target) {
            this.closeSidePanel();
            this.currentWorkCode = target.code;
            this.currentDivision = division !== undefined
                ? Math.max(target.minDivision, Math.min(division, target.totalDivisions))
                : target.minDivision;
            this.currentSection = section ?? null;
        }
    }

    setDivision(division: number, section?: string | number | null) {
        this.closeSidePanel();
        const work = this.currentWork;
        this.currentDivision = Math.max(work.minDivision, Math.min(division, work.totalDivisions));
        this.currentSection = section ?? null;
    }

    nextDivision() {
        const work = this.currentWork;
        if (this.currentDivision < work.totalDivisions) {
            this.closeSidePanel();
            this.currentDivision += 1;
            this.currentSection = null;
        }
    }

    prevDivision() {
        const work = this.currentWork;
        if (this.currentDivision > work.minDivision) {
            this.closeSidePanel();
            this.currentDivision -= 1;
            this.currentSection = null;
        }
    }

    closeSidePanel() {
        this.isSidePanelOpen = false;
        this.selectedToken = null;
        this.selectedCitation = null;
        this.selectedScriptureCitation = null;
    }

    openToken(token: TextToken, citation: string) {
        this.selectedToken = token;
        this.selectedCitation = citation;
        this.activeSidePanelTab = 'lexeme';
        this.isSidePanelOpen = true;
    }
}

export const readerState = new ReaderState();
```
