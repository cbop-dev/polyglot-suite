## 1. Objective & Scope

This specification details the refactoring of the presentation tier in both `apps/ancient-corpora-reader` and the shared component library `@polyglot/ui`.

With the deletion of legacy standalone reader apps (`philo-reader`, `josephus-reader`, `af-reader`), **`ParallelColumnView.svelte` is refactored directly to accept generic text versions and dynamic language visibility sets**, permanently eliminating hardcoded language columns and edition strings.

### 2.2 Dynamic Column Layout & Responsive Grid
Instead of static conditionals checking `hasGreek`, `hasLatin`, and `hasEng`, `ParallelColumnView` computes active columns dynamically:

```ts
// Filter active versions based on visibleLanguages
let activeVersions = $derived.by(() => {
    if (!visibleLanguages || visibleLanguages.size === 0) return textVersions;
    return textVersions.filter(
        (v) => visibleLanguages.has(v.language) || visibleLanguages.has(v.versionId)
    );
});

// Dynamic responsive grid classes based on active column count
let gridColsClass = $derived.by(() => {
    if (viewMode === 'stacked' || activeVersions.length <= 1) return 'grid-cols-1';
    const count = activeVersions.length;
    if (gridBreakpoint === 'md') {
        if (count >= 3) return 'grid-cols-1 md:grid-cols-3';
        if (count === 2) return 'grid-cols-1 md:grid-cols-2';
    } else {
        if (count >= 3) return 'grid-cols-1 lg:grid-cols-3';
        if (count === 2) return 'grid-cols-1 lg:grid-cols-2';
    }
    return 'grid-cols-1';
});
```

### 2.3 Dynamic Section Row Synchronization & Progressive Batch Rendering

#### The Large Page Performance Problem:
In philosophical and historical literature (especially Philo and Josephus), a single "division" or treatise (e.g. Philo's *De Opificio Mundi*) often contains **150 to 300+ sections**, encompassing **15,000 to 30,000 interactive word token elements**.
* If Svelte attempts to construct, hydrate, and mount 25,000 `<button>` elements with highlight store subscriptions in a single synchronous render frame, the browser main thread freezes for **1.5 to 3.5 seconds** (high Total Blocking Time).
* Even with `content-visibility: auto` skipping paint, the DOM creation and reactivity registration freeze the UI.

#### The Solution: Progressive Batch Rendering
`ParallelColumnView.svelte` incorporates a built-in **Progressive Batch Rendering Engine**:
1. **Instant First Paint (Batch 1)**: Renders the first 25 sections immediately (< 50ms) so the viewport is interactive and readable instantly.
2. **Idle Callback Batch Extension**: While the user is reading or idle, incrementally appends subsequent batches (25 sections at a time) using `requestIdleCallback` (with `requestAnimationFrame` fallback).
3. **Scroll-Driven Sentinel (`IntersectionObserver`)**: A sentinel element at the bottom of the rendered slice triggers the immediate rendering of the next batch if the user scrolls faster than the idle callback loop.
4. **Target Section Jump Handling**: When navigating directly to a deep section (e.g. section 120 via URL hash or picker), the initial batch size is dynamically adjusted: `Math.max(25, targetIndex + 15)`, guaranteeing the target section is rendered in Batch 1 and immediately visible without delay.

```ts
// Progressive Batch Rendering State in ParallelColumnView
const INITIAL_BATCH_SIZE = 25;
const BATCH_STEP_SIZE = 25;

let renderedCount = $state<number>(INITIAL_BATCH_SIZE);
let renderedKeys = $derived(sectionKeys.slice(0, renderedCount));
let isFullyRendered = $derived(renderedCount >= sectionKeys.length);

let idleHandle: any = null;

function scheduleNextBatch() {
    if (isFullyRendered || typeof window === 'undefined') return;
    
    const run = (deadline?: IdleDeadline) => {
        // While idle time remains, advance rendered batches
        while ((!deadline || deadline.timeRemaining() > 5) && renderedCount < sectionKeys.length) {
            renderedCount = Math.min(sectionKeys.length, renderedCount + BATCH_STEP_SIZE);
        }
        if (renderedCount < sectionKeys.length) {
            scheduleNextBatch();
        }
    };

    if ('requestIdleCallback' in window) {
        idleHandle = (window as any).requestIdleCallback(run, { timeout: 150 });
    } else {
        idleHandle = setTimeout(run, 50);
    }
}

// Reset batching on division or locus change
$effect(() => {
    // Track dependency on sectionKeys
    if (sectionKeys.length > 0) {
        if (targetSection) {
            const idx = sectionKeys.findIndex(k => String(k) === String(targetSection));
            renderedCount = idx >= 0 ? Math.min(sectionKeys.length, Math.max(INITIAL_BATCH_SIZE, idx + 15)) : INITIAL_BATCH_SIZE;
        } else {
            renderedCount = INITIAL_BATCH_SIZE;
        }
        scheduleNextBatch();
    }
    return () => {
        if (idleHandle && typeof window !== 'undefined') {
            if ('cancelIdleCallback' in window) (window as any).cancelIdleCallback(idleHandle);
            else clearTimeout(idleHandle);
        }
    };
});
```

#### Synchronized Render Loop:
```svelte
{#each renderedKeys as secKey (secKey)}
    {@const unitsByVer = getUnitsForSection(secKey)}
    
    <div id={rowId ? rowId(secKey) : `section-${secKey}`}
         class="section-row {viewMode === 'stacked' ? 'flex flex-col gap-4' : `grid ${gridColsClass} gap-6`} py-5">
        
        {#each activeVersions as ver (ver.versionId)}
            {@const unit = unitsByVer[ver.versionId]}
            <div class="column-version {ver.fontClass}" dir={ver.direction}>
                <!-- Section Badge in primary leading column -->
                {#if ver === activeVersions[0]}
                    {#if sectionBadge}
                        {@render sectionBadge({ secKey, units: unitsByVer })}
                    {:else}
                        <span class="section-badge font-bold text-xs mr-2">§ {secKey}</span>
                    {/if}
                {/if}

                <!-- Token Word Rendering with Dynamic Font & RTL Handling -->
                {#if unit?.tokens && unit.tokens.length > 0}
                    {#each unit.tokens as tok}
                        <!-- Render Token with highlightStore & onWordClick -->
                    {/each}
                {:else if unit?.text_content}
                    <span>{unit.text_content}</span>
                {:else}
                    <span class="text-sm italic text-ink-soft">—</span>
                {/if}
            </div>
        {/each}
    </div>
{/each}

<!-- Sentinel for fast-scroll progressive trigger -->
{#if !isFullyRendered}
    <div class="h-8 flex items-center justify-center text-xs text-ink-soft select-none py-2" aria-hidden="true">
        <span class="animate-pulse">Loading sections... ({renderedCount} / {sectionKeys.length})</span>
    </div>
{/if}
```

## 3. Refactoring `CorpusColumnView.svelte`

`CorpusColumnView.svelte` becomes a concise container bridging `readerState` and `ParallelColumnView`:

```svelte
<script lang="ts">
    import { ParallelColumnView } from '@polyglot/ui';
    import { readerState } from '../stores/readerState.svelte.js';
    import { highlightStore } from '../stores/highlightStore.svelte.js';
    import type { TextVersion } from '../stores/types.js';

    interface Props {
        textVersions: TextVersion[];
    }

    let { textVersions = [] }: Props = $props();
    let cfg = $derived(readerState.currentCorpusConfig);
</script>

<ParallelColumnView
    {textVersions}
    visibleLanguages={readerState.visibleLanguages}
    viewMode={readerState.viewMode}
    {highlightStore}
    onWordClick={(tok, citation, ver) => readerState.openToken(tok, citation)}
    onBiblicalCitationClick={(code, ch, v, disp, name) => 
        readerState.openScriptureCitation(code, ch, v, disp, name)}
>
    {#snippet sectionBadge({ secKey, units })}
        {@const milestone = cfg.milestoneScheme}
        {@const primaryUnit = Object.values(units)[0]}
        
        <div class="inline-flex items-center gap-1.5 mr-2 mb-1 select-none">
            <button type="button" class="text-xs font-mono font-bold text-blue-600 hover:underline">
                {secKey}
            </button>
            {#if milestone && primaryUnit?.milestone_ref}
                <span class="rounded px-1 py-0.2 text-[10px] font-mono {milestone.badgeColorClass}">
                    {milestone.badgePrefix} {primaryUnit.milestone_ref}
                </span>
            {/if}
        </div>
    {/snippet}
</ParallelColumnView>
```
